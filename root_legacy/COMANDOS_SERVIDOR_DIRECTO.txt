PASOS COMPLETOS:
----------------

1. Conectarse al servidor (ya estás conectado):
   ssh atlantareciclajes@ssh.pythonanywhere.com

2. Ir al directorio del proyecto:
   cd /home/atlantareciclajes/apps/egarage/current

3. Verificar que estás en el lugar correcto:
   pwd
   # Debe mostrar: /home/atlantareciclajes/apps/egarage/current
   
   ls manage.py
   # Debe mostrar: manage.py

4. Activar virtualenv:
   source ~/.virtualenvs/venv_egarage310/bin/activate

5. Verificar que Python está correcto:
   which python
   # Debe mostrar: /home/atlantareciclajes/.virtualenvs/venv_egarage310/bin/python

6. Ahora ejecutar los comandos:
   python manage.py diagnosticar_autenticacion --usuario testuser_usa --fix
   python manage.py listar_usuarios_credenciales --usuario testuser_usa

from django.contrib.auth import get_user_model, authenticate
from django.db import transaction
from datetime import datetime, timedelta
from taller.models import Empresa, Suscripcion

User = get_user_model()

username = "testuser_usa"
password = "TestUSA2025!"

print("=" * 80)
print(f"🔍 DIAGNÓSTICO Y CORRECCIÓN: {username}")
print("=" * 80)
print()

# Verificar si existe
try:
    user = User.objects.get(username=username)
    print(f"✅ Usuario encontrado: {user.username}")
    print(f"   Email: {user.email}")
    print(f"   Activo: {user.is_active}")
    print()
    
    # Verificar contraseña
    password_correcta = user.check_password(password)
    print(f"🔑 Verificación de contraseña:")
    print(f"   Esperada: {password}")
    print(f"   Estado: {'✅ CORRECTA' if password_correcta else '❌ INCORRECTA'}")
    print()
    
    # Si no es correcta, resetear
    if not password_correcta:
        print("🔧 Reseteando contraseña...")
        with transaction.atomic():
            user.set_password(password)
            user.is_active = True
            user.save()
        user.refresh_from_db()
        if user.check_password(password):
            print("✅ Contraseña reseteada exitosamente")
        else:
            print("❌ ERROR: No se pudo resetear la contraseña")
        print()
    
    # Verificar empresa
    try:
        empresa = user.empresa
        print(f"✅ Empresa: {empresa.nombre_taller}")
        print(f"   País: {empresa.pais}")
        print(f"   Moneda: {getattr(empresa, 'moneda', 'USD')}")
    except AttributeError:
        print("❌ Usuario no tiene empresa. Creando...")
        with transaction.atomic():
            empresa = Empresa.objects.create(
                user=user,
                nombre_taller="Taller de testuser_usa",
                pais="US",
                telefono="+15551234567",
                moneda="USD"
            )
        print(f"✅ Empresa creada: {empresa.nombre_taller}")
        print()
    
    # Verificar suscripción
    try:
        suscripcion = user.suscripcion
        print(f"✅ Suscripción: {suscripcion.tipo}")
        print(f"   Activa: {'✅ Sí' if suscripcion.activa else '❌ No'}")
        if suscripcion.fecha_fin:
            vencida = suscripcion.fecha_fin < datetime.now().date()
            print(f"   Válida hasta: {suscripcion.fecha_fin} ({'✅ Vigente' if not vencida else '❌ Vencida'})")
        
        if not suscripcion.activa or (suscripcion.fecha_fin and suscripcion.fecha_fin < datetime.now().date()):
            print("   🔧 Renovando suscripción...")
            with transaction.atomic():
                suscripcion.activa = True
                suscripcion.fecha_fin = (datetime.now() + timedelta(days=30)).date()
                suscripcion.save()
            print(f"   ✅ Suscripción renovada hasta {suscripcion.fecha_fin}")
    except AttributeError:
        print("❌ Usuario no tiene suscripción. Creando...")
        with transaction.atomic():
            suscripcion = Suscripcion.objects.create(
                user=user,
                tipo="trial",
                activa=True,
                fecha_inicio=datetime.now().date(),
                fecha_fin=(datetime.now() + timedelta(days=30)).date()
            )
        print(f"✅ Suscripción creada hasta {suscripcion.fecha_fin}")
        print()
    
    # Verificar autenticación
    print("🔐 Verificación de Autenticación Django:")
    auth_user = authenticate(username=username, password=password)
    if auth_user:
        print("   ✅ Autenticación Django: EXITOSA")
    else:
        print("   ❌ Autenticación Django: FALLIDA")
        if not user.is_active:
            print("   Causa: Usuario inactivo")
        else:
            print("   Causa: Desconocida")
    print()
    
    # Resumen final
    print("=" * 80)
    print("✅ RESUMEN")
    print("=" * 80)
    print(f"Usuario: {username}")
    print(f"Contraseña: {password}")
    print("URL: https://www.egarage.cl/us/accounts/login/")
    print("=" * 80)

from django.contrib.auth import get_user_model
from taller.models import Empresa, Suscripcion

User = get_user_model()

print("=" * 80)
print("🔍 TODOS LOS USUARIOS DEL SISTEMA")
print("=" * 80)
print()

usuarios = User.objects.all().order_by('username')
print(f"Total: {usuarios.count()} usuarios")
print()

credenciales_conocidas = {
    'testuser_usa': 'TestUSA2025!',
    'test_usa': 'test1234',
    'test_usa_pago': 'test1234',
    'test_chile': 'test1234',
    'test_chile_pago': 'test1234',
    'testuser_cl': 'test123',
    'admin_chile': 'admin123',
    'admin_usa': 'admin123',
    'admin': 'admin123',
}

for user in usuarios:
    estado = "✅ Activo" if user.is_active else "❌ Inactivo"
    tipo = "👑 Superuser" if user.is_superuser else ("🔧 Staff" if user.is_staff else "👤 Normal")
    
    credencial = None
    if user.username in credenciales_conocidas:
        password_esperada = credenciales_conocidas[user.username]
        password_valida = user.check_password(password_esperada)
        credencial = f"{user.username} / {password_esperada} ({'✅' if password_valida else '❌'})"
    
    print(f"{tipo} {user.username}")
    print(f"   Email: {user.email or 'Sin email'}")
    print(f"   Estado: {estado}")
    
    if credencial:
        print(f"   🔑 Credencial: {credencial}")
    
    try:
        empresa = user.empresa
        print(f"   Empresa: {empresa.nombre_taller}, País: {empresa.pais}")
    except:
        print("   ⚠️  Sin empresa")
    
    try:
        suscripcion = user.suscripcion
        print(f"   Suscripción: {suscripcion.tipo}, Activa: {suscripcion.activa}")
    except:
        print("   ⚠️  Sin suscripción")
    
    print()
