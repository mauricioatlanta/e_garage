import re

template_path = 'taller/templates/taller/us/en/vehiculos/crear_vehiculo.html'

with open(template_path, 'r', encoding='utf-8') as f:
    lines = f.readlines()

# Eliminar líneas que contienen los indicadores
new_lines = []
skip = False
for i, line in enumerate(lines):
    # Detectar inicio del bloque de la flor
    if '<!-- FLOR INDICADORA' in line:
        skip = True
        continue
    # Detectar inicio del bloque de la estrella
    if '<!-- ESTRELLA INDICADORA' in line:
        skip = True
        continue
    # Detectar fin del bloque (div cerrado antes de max-w-4xl)
    if skip and '</div>' in line:
        # Verificar si la siguiente línea es el div principal
        if i + 1 < len(lines) and 'max-w-4xl' in lines[i + 1]:
            skip = False
            continue
        else:
            skip = False
            continue
    # Detectar y eliminar animaciones CSS
    if '@keyframes florPulse' in line:
        # Eliminar hasta encontrar el cierre }
        skip_animation = True
        continue
    if '@keyframes starPulse' in line:
        skip_animation = True
        continue
    if skip_animation:
        if line.strip() == '}':
            skip_animation = False
        continue
    # Saltar líneas dentro de los bloques
    if skip:
        continue
    # Mantener la línea
    new_lines.append(line)

# Escribir el archivo sin los indicadores
with open(template_path, 'w', encoding='utf-8') as f:
    f.writelines(new_lines)

print("✓ Indicadores eliminados")

from django.urls import reverse

# Verificar URLs de autocomplete de clientes
urls_to_check = [
    'usa:vehiculos:cliente_autocomplete',
    'usa:autocomplete:cliente',
    'usa:taller:vehiculos:cliente_autocomplete',
]

print("=== Verificando URLs de autocomplete de clientes ===")
for url_name in urls_to_check:
    try:
        url = reverse(url_name)
        print(f"✓ {url_name} → {url}")
    except Exception as e:
        print(f"✗ {url_name} → {str(e)[:60]}")

# Verificar qué namespace tiene el autocomplete
from django.urls import get_resolver
resolver = get_resolver()
if 'usa' in resolver.namespace_dict:
    usa_resolver = resolver.namespace_dict['usa'][1]
    if 'autocomplete' in usa_resolver.namespace_dict:
        print("\n✓ Namespace 'usa:autocomplete' encontrado")
        autocomplete_resolver = usa_resolver.namespace_dict['autocomplete'][1]
        print(f"  URLs disponibles: {len(autocomplete_resolver.url_patterns)}")
        for pattern in autocomplete_resolver.url_patterns[:5]:
            if hasattr(pattern, 'name') and pattern.name:
                print(f"    - {pattern.name}")
exit()
