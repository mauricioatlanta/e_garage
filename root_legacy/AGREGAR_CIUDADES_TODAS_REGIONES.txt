print("🔧 Agregando ciudades a todas las regiones...\n")
print("=" * 70)

total_ciudades_creadas = 0
total_ciudades_existentes = 0

for nombre_region, ciudades in regiones_ciudades.items():
    # Buscar región (puede estar con o sin acento)
    region = TallerRegion.objects.filter(nombre__iexact=nombre_region).first()
    
    if not region:
        # Intentar variaciones sin acentos
        nombre_sin_acentos = nombre_region.replace("á", "a").replace("é", "e").replace("í", "i").replace("ó", "o").replace("ú", "u").replace("ñ", "n")
        region = TallerRegion.objects.filter(nombre__iexact=nombre_sin_acentos).first()
    
    if not region:
        print(f"⚠️  Región no encontrada: {nombre_region}")
        continue
    
    print(f"\n📍 Región: {region.nombre} (ID: {region.id})")
    ciudades_creadas = 0
    ciudades_existentes = 0
    
    for ciudad_nombre in ciudades:
        ciudad, ciudad_creada = TallerCiudad.objects.get_or_create(
            nombre=ciudad_nombre,
            region=region
        )
        if ciudad_creada:
            print(f"   ✅ {ciudad_nombre}")
            ciudades_creadas += 1
            total_ciudades_creadas += 1
        else:
            ciudades_existentes += 1
            total_ciudades_existentes += 1
    
    print(f"   📊 Creadas: {ciudades_creadas}, Existentes: {ciudades_existentes}, Total en región: {TallerCiudad.objects.filter(region=region).count()}")

print()
print("=" * 70)
print(f"📊 Resumen General:")
print(f"   ✅ Total ciudades creadas: {total_ciudades_creadas}")
print(f"   ℹ️  Total ciudades existentes: {total_ciudades_existentes}")
print(f"   📈 Total ciudades en BD: {TallerCiudad.objects.count()}")
print("=" * 70)
print("\n✅ Proceso completado")
