import re
import os

print("=" * 60)
print("APLICANDO CAMBIOS EN ARCHIVOS")
print("=" * 60)

# 1. Modificar taller/servicios/views.py
views_file = 'taller/servicios/views.py'
if not os.path.exists(views_file):
    print(f"❌ No se encuentra: {views_file}")
    exit(1)

print(f"\n📝 Modificando {views_file}...")

with open(views_file, 'r', encoding='utf-8') as f:
    content = f.read()

# Backup
with open(views_file + '.backup', 'w', encoding='utf-8') as f:
    f.write(content)
print("✅ Backup creado")

# Cambio 1: Filtrado de servicios por país
old_pattern = r'(# Obtener servicios con filtros básicos.*?servicios_list = list\(servicios_qs\))'
new_services = '''    # Obtener servicios con filtros básicos
    # Primero filtrar por país (importante para mostrar servicios correctos)
    servicios = Servicio.objects.filter(categoria__country=country_code)
    
    # Si el usuario tiene empresa, priorizar servicios de su empresa
    if empresa:
        servicios_empresa = servicios.filter(empresa=empresa)
        if servicios_empresa.exists():
            servicios = servicios_empresa
        # Si no tiene servicios en su empresa, mostrar todos del país
        # (ya está filtrado por país arriba)
    
    servicios_qs = (
        servicios.select_related("categoria", "subcategoria")
        .prefetch_related("names", "categoria__names", "subcategoria__names")
        .order_by("nombre")
    )
    if language:
        servicios_qs = servicios_qs.filter(names__language=language).distinct()
    servicios_list = list(servicios_qs)'''

if re.search(old_pattern, content, re.DOTALL):
    content = re.sub(old_pattern, new_services, content, flags=re.DOTALL)
    print("✅ Filtrado de servicios actualizado")
else:
    print("⚠️ No se encontró el patrón de servicios (puede que ya esté actualizado)")

# Cambio 2: Categorías
old_cat_pattern = r'(categorias = list\(\s+CategoriaServicio\.objects\.filter\(\s+country=country_code,\s+names__language=language,\s+names__is_default=True,\s+\)\s+\.prefetch_related\("names"\)\s+\.order_by\("code"\)\s+\.distinct\(\)\s+\)\s+for categoria in categorias:\s+categoria\.label_localizado = categoria\.get_label\(language\))'
new_categorias = '''    # Obtener categorías del país, asegurando que tengan nombres en el idioma correcto
    categorias_qs = CategoriaServicio.objects.filter(
        country=country_code
    ).prefetch_related("names").order_by("code").distinct()
    
    categorias = []
    for categoria in categorias_qs:
        # Verificar que tenga nombre en el idioma solicitado
        nombre = categoria.get_label(language)
        if nombre and nombre != categoria.code:  # Si tiene nombre válido
            categoria.label_localizado = nombre
            categorias.append(categoria)'''

if re.search(old_cat_pattern, content, re.DOTALL):
    content = re.sub(old_cat_pattern, new_categorias, content, flags=re.DOTALL)
    print("✅ Filtrado de categorías actualizado")
else:
    print("⚠️ No se encontró el patrón de categorías (puede que ya esté actualizado)")

# Cambio 3: Subcategorías
old_sub_pattern = r'(subcategorias = list\(\s+SubcategoriaServicio\.objects\.filter\(\s+country=country_code,\s+names__language=language,\s+names__is_default=True,\s+\)\s+\.prefetch_related\("names"\)\s+\.order_by\("code"\)\s+\.distinct\(\)\s+\)\s+for subcategoria in subcategorias:\s+subcategoria\.label_localizado = subcategoria\.get_label\(language\))'
new_subcategorias = '''    # Obtener subcategorías del país, asegurando que tengan nombres en el idioma correcto
    subcategorias_qs = SubcategoriaServicio.objects.filter(
        country=country_code
    ).prefetch_related("names").order_by("code").distinct()
    
    subcategorias = []
    for subcategoria in subcategorias_qs:
        # Verificar que tenga nombre en el idioma solicitado
        nombre = subcategoria.get_label(language)
        if nombre and nombre != subcategoria.code:  # Si tiene nombre válido
            subcategoria.label_localizado = nombre
            subcategorias.append(subcategoria)'''

if re.search(old_sub_pattern, content, re.DOTALL):
    content = re.sub(old_sub_pattern, new_subcategorias, content, flags=re.DOTALL)
    print("✅ Filtrado de subcategorías actualizado")
else:
    print("⚠️ No se encontró el patrón de subcategorías (puede que ya esté actualizado)")

# Cambio 4: servicios_por_categoria
if 'if servicio.categoria:' in content and 'servicio.categoria.country == country_code' not in content:
    content = content.replace(
        'if servicio.categoria:',
        'if servicio.categoria and servicio.categoria.country == country_code:'
    )
    print("✅ Validación de país en servicios_por_categoria actualizada")

# Guardar
with open(views_file, 'w', encoding='utf-8') as f:
    f.write(content)
print(f"✅ {views_file} guardado\n")

# 2. Modificar template
template_file = 'templates/taller/us/en/servicios/servicios_menu.html'
if not os.path.exists(template_file):
    print(f"❌ No se encuentra: {template_file}")
    exit(1)

print(f"📝 Modificando {template_file}...")

with open(template_file, 'r', encoding='utf-8') as f:
    template_content = f.read()

# Backup
with open(template_file + '.backup', 'w', encoding='utf-8') as f:
    f.write(template_content)
print("✅ Backup creado")

# Reemplazos en template
replacements = [
    (r'\{\{ servicio\.nombre \}\}', '{{ servicio.nombre_localizado|default:servicio.nombre }}'),
    (r'\{\{ servicio\.subcategoria\.get_label \}\}', '{{ servicio.subcategoria_label|default:servicio.subcategoria.get_label }}'),
    (r'\{\{ categoria\.get_label \}\}', '{{ categoria.label_localizado|default:categoria.get_label }}'),
    (r'\{\{ subcategoria\.get_label \}\}', '{{ subcategoria.label_localizado|default:subcategoria.get_label }}'),
    (r'\{\{ servicio\.categoria\.get_label\|default:\'\' \}\}', '{{ servicio.categoria_label|default:servicio.categoria.get_label|default:\'\' }}'),
    (r'\{\{ servicio\.subcategoria\.get_label\|default:\'\' \}\}', '{{ servicio.subcategoria_label|default:servicio.subcategoria.get_label|default:\'\' }}'),
    (r'\{\{ servicio\.nombre\|default:\'\' \}\}', '{{ servicio.nombre_localizado|default:servicio.nombre|default:\'\' }}'),
    (r'\{\{ servicios_cat\.count \}\}', '{{ servicios_cat|length }}'),
]

changes_made = 0
for pattern, replacement in replacements:
    if re.search(pattern, template_content):
        template_content = re.sub(pattern, replacement, template_content)
        changes_made += 1
        print(f"✅ Reemplazo aplicado: {pattern[:40]}...")

if changes_made > 0:
    print(f"✅ {changes_made} reemplazos realizados en template")
else:
    print("⚠️ No se encontraron patrones para reemplazar (puede que ya estén actualizados)")

# Guardar
with open(template_file, 'w', encoding='utf-8') as f:
    f.write(template_content)
print(f"✅ {template_file} guardado\n")
