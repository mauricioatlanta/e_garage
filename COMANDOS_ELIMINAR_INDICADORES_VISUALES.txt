# 1. Eliminar los divs de la flor y estrella (líneas 493-542)
# Usar sed para eliminar desde "<!-- FLOR INDICADORA" hasta antes de "<div class="max-w-4xl"
sed -i '/<!-- FLOR INDICADORA/,/<\/div>/d' taller/templates/taller/us/en/vehiculos/crear_vehiculo.html

# 2. Eliminar las animaciones CSS florPulse y starPulse
sed -i '/@keyframes florPulse/,/^}$/d' taller/templates/taller/us/en/vehiculos/crear_vehiculo.html
sed -i '/@keyframes starPulse/,/^}$/d' taller/templates/taller/us/en/vehiculos/crear_vehiculo.html

# 3. Verificar que se eliminaron
grep -n "🌸\|⭐\|florPulse\|starPulse\|FLOR INDICADORA\|ESTRELLA INDICADORA" \
   taller/templates/taller/us/en/vehiculos/crear_vehiculo.html
# No debe mostrar nada

# 4. Limpiar caché
find . -type d -name "__pycache__" -exec rm -r {} + 2>/dev/null
find . -name "*.pyc" -delete
