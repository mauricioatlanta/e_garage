from django.template.loader import get_template
from taller.vehiculos.views_fbv import _vehicle_template

template_name = _vehicle_template("crear", "US")
print(f"Template esperado: {template_name}")

try:
    template = get_template(template_name)
    print(f"✓ Template cargado: {template.origin}")
    
    # Leer el archivo directamente
    import os
    template_path = str(template.origin)
    if os.path.exists(template_path):
        size = os.path.getsize(template_path)
        print(f"✓ Tamaño: {size} bytes ({size/1024:.1f} KB)")
        
        with open(template_path, 'r', encoding='utf-8') as f:
            content = f.read(1000)  # Leer primeros 1000 caracteres
            if 'Create Vehicle' in content or '🌸' in content or '⭐' in content:
                print("✓ Template contiene el contenido esperado")
            else:
                print("✗ Template NO contiene el contenido esperado")
                print(f"  Primeros 200 caracteres: {content[:200]}")
    else:
        print(f"✗ Archivo no existe: {template_path}")
except Exception as e:
    print(f"✗ Error: {e}")
    import traceback
    traceback.print_exc()
exit()
