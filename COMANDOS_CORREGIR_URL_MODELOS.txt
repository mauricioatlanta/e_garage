from django.urls import reverse

urls_to_check = [
    'usa:vehiculos:ajax_agregar_motor',
    'usa:vehiculos:ajax_agregar_caja',
    'usa:vehiculos:ajax_modelos_por_marca',
    'usa:vehiculos:ajax_modelos_por_marca_anio',
    'usa:vehiculos:ajax_motores_por_modelo',
    'usa:vehiculos:ajax_cajas_por_modelo',
    'usa:vehiculos:api_marcas',
]

print("=== Verificando URLs del template ===")
all_ok = True
for url_name in urls_to_check:
    try:
        url = reverse(url_name)
        print(f"✓ {url_name}")
    except Exception as e:
        print(f"✗ {url_name} → {str(e)[:60]}")
        all_ok = False

if all_ok:
    print("\n✓ Todas las URLs están disponibles")
else:
    print("\n✗ Algunas URLs no están disponibles")
exit()
