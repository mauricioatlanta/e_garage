total_servicios_creados = 0
total_servicios_existentes = 0

# Obtener todas las subcategorías de Chile
subcategorias = SubcategoriaServicio.objects.filter(country='CL').select_related('categoria')

print(f"\n📋 Subcategorías encontradas: {subcategorias.count()}")

for empresa in empresas:
    print(f"\n{'=' * 70}")
    if empresa:
        print(f"🏢 Procesando empresa: {empresa.nombre if hasattr(empresa, 'nombre') else str(empresa)}")
    else:
        print(f"🏢 Procesando sin empresa (puede fallar)")
    
    for subcategoria in subcategorias:
        servicios_lista = servicios_por_subcategoria.get(subcategoria.code, [])
        
        if not servicios_lista:
            continue
        
        for servicio_nombre in servicios_lista:
            try:
                servicio, created = Servicio.objects.get_or_create(
                    empresa=empresa,
                    nombre=servicio_nombre,
                    categoria=subcategoria.categoria,
                    subcategoria=subcategoria,
                    defaults={
                        'empresa': empresa,
                        'nombre': servicio_nombre,
                        'categoria': subcategoria.categoria,
                        'subcategoria': subcategoria
                    }
                )
                
                if created:
                    # Crear nombre en español
                    ServicioName.objects.get_or_create(
                        servicio=servicio,
                        language='es',
                        is_default=True,
                        defaults={
                            'label': servicio_nombre,
                            'aliases': []
                        }
                    )
                    total_servicios_creados += 1
                    print(f"   ✅ Servicio creado: {servicio_nombre}")
                else:
                    total_servicios_existentes += 1
                    
            except Exception as e:
                print(f"   ❌ Error creando servicio '{servicio_nombre}': {e}")
                continue

print("\n" + "=" * 70)
print("✅ RESUMEN FINAL")
print("=" * 70)
print(f"   ✅ Servicios creados: {total_servicios_creados}")

print(f"   📈 Total servicios en BD: {Servicio.objects.count()}")
print("=" * 70)
print("\n✅ Proceso completado. Recarga la página para ver los servicios.")
