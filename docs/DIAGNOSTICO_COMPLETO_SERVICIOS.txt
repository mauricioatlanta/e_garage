import os
import django

os.environ.setdefault('DJANGO_SETTINGS_MODULE', 'gestion_taller.settings')

try:
    django.setup()
except Exception as e:
    print(f"Warning: {e}")

from taller.models import Empresa
from taller.servicios.models import (
    CategoriaServicio, CategoriaServicioName,
    Servicio, ServicioName,
    SubcategoriaServicio, SubcategoriaServicioName,
)

print("=" * 70)
print("DIAGNÓSTICO COMPLETO: SERVICIOS US")
print("=" * 70)

country_code = "US"
language = "en"

# 1. Verificar categorías US
print(f"\n📂 CATEGORÍAS US:")
categorias_us = CategoriaServicio.objects.filter(country=country_code)
print(f"   Total categorías US: {categorias_us.count()}")

for cat in categorias_us:
    nombres = cat.names.filter(language=language, is_default=True)
    nombre_en = nombres.first().label if nombres.exists() else None
    print(f"   - {cat.code}: {nombre_en or 'SIN NOMBRE EN'}")
    
    # Verificar si tiene nombres en otros idiomas
    otros_nombres = cat.names.exclude(language=language)
    if otros_nombres.exists():
        print(f"     ⚠️ Tiene nombres en otros idiomas: {', '.join([n.language for n in otros_nombres])}")

# 2. Verificar subcategorías US
print(f"\n📁 SUBCATEGORÍAS US:")
subcategorias_us = SubcategoriaServicio.objects.filter(country=country_code)
print(f"   Total subcategorías US: {subcategorias_us.count()}")

for sub in subcategorias_us[:10]:  # Mostrar primeras 10
    nombres = sub.names.filter(language=language, is_default=True)
    nombre_en = nombres.first().label if nombres.exists() else None
    print(f"   - {sub.code}: {nombre_en or 'SIN NOMBRE EN'} (Categoría: {sub.categoria.code})")

# 3. Verificar servicios US
print(f"\n🔧 SERVICIOS US:")
servicios_us = Servicio.objects.filter(categoria__country=country_code)
print(f"   Total servicios con categoría US: {servicios_us.count()}")

# Verificar servicios con nombres en inglés
servicios_con_en = servicios_us.filter(
    names__language=language,
    names__is_default=True
).distinct()
print(f"   Servicios con nombre EN: {servicios_con_en.count()}")

# Mostrar algunos servicios
print(f"\n   Ejemplos de servicios:")
for serv in servicios_con_en[:5]:
    nombre_en = serv.names.filter(language=language, is_default=True).first()
    print(f"   - {serv.nombre}")
    print(f"     Nombre EN: {nombre_en.label if nombre_en else 'SIN NOMBRE'}")
    print(f"     Categoría: {serv.categoria.code} ({serv.categoria.country})")
    print(f"     Subcategoría: {serv.subcategoria.code if serv.subcategoria else 'N/A'}")
    print(f"     Empresa: {serv.empresa.nombre_taller or serv.empresa.empresa or f'Empresa {serv.empresa.id}'}")

# 4. Verificar filtrado como en la vista
print(f"\n🔍 FILTRADO COMO EN LA VISTA:")
servicios_filtrados = Servicio.objects.filter(categoria__country=country_code)
servicios_filtrados = servicios_filtrados.select_related("categoria", "subcategoria").prefetch_related(
    "names", "categoria__names", "subcategoria__names"
).order_by("nombre")

if language:
    servicios_filtrados = servicios_filtrados.filter(names__language=language).distinct()

servicios_list = list(servicios_filtrados)
print(f"   Servicios después del filtrado: {len(servicios_list)}")

# 5. Verificar categorías filtradas como en la vista
print(f"\n📋 CATEGORÍAS FILTRADAS:")
categorias_filtradas = CategoriaServicio.objects.filter(
    country=country_code
).prefetch_related("names").order_by("code").distinct()

categorias_validas = []
for categoria in categorias_filtradas:
    nombre = categoria.get_label(language)
    if nombre and nombre != categoria.code:
        categorias_validas.append(categoria)
        print(f"   ✅ {categoria.code}: {nombre}")
    else:
        print(f"   ❌ {categoria.code}: SIN NOMBRE VÁLIDO")

print(f"\n   Total categorías válidas: {len(categorias_validas)}")

# 6. Verificar servicios por categoría
print(f"\n📊 SERVICIOS POR CATEGORÍA:")
from collections import defaultdict
servicios_por_categoria = defaultdict(list)
for servicio in servicios_list:
    if servicio.categoria and servicio.categoria.country == country_code:
        servicios_por_categoria[servicio.categoria].append(servicio)

for categoria, servicios_cat in servicios_por_categoria.items():
    nombre_cat = categoria.get_label(language)
    print(f"   - {categoria.code} ({nombre_cat}): {len(servicios_cat)} servicios")

print(f"\n✅ DIAGNÓSTICO COMPLETADO")
print(f"   Si los servicios no aparecen, puede ser:")
print(f"   1. El usuario no está autenticado")
print(f"   2. El usuario tiene otra empresa")
print(f"   3. Hay un problema con el template")
