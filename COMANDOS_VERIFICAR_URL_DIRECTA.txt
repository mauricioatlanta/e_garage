from django.urls import reverse, get_resolver

# Intentar todas las combinaciones posibles
urls_to_test = [
    'usa:vehiculos:modelos_por_marca_api',
    'usa:taller:vehiculos:modelos_por_marca_api',
    'vehiculos:modelos_por_marca_api',
    'taller:vehiculos:modelos_por_marca_api',
]

print("=== Probando URLs ===")
for url_name in urls_to_test:
    try:
        url = reverse(url_name)
        print(f"✓ {url_name} → {url}")
    except Exception as e:
        print(f"✗ {url_name} → {str(e)[:80]}")

# Verificar qué URLs están realmente disponibles en vehiculos
print("\n=== URLs disponibles en vehiculos ===")
try:
    resolver = get_resolver()
    # Buscar usa namespace
    if 'usa' in resolver.namespace_dict:
        usa_resolver = resolver.namespace_dict['usa'][1]
        print("✓ Namespace 'usa' encontrado")
        
        # Buscar vehiculos dentro de usa
        if 'vehiculos' in usa_resolver.namespace_dict:
            vehiculos_resolver = usa_resolver.namespace_dict['vehiculos'][1]
            print("✓ Namespace 'usa:vehiculos' encontrado")
            print(f"  Total de URLs: {len(vehiculos_resolver.url_patterns)}")
            print("\n  URLs disponibles:")
            for pattern in vehiculos_resolver.url_patterns:
                if hasattr(pattern, 'name') and pattern.name:
                    print(f"    - {pattern.name}")
        else:
            print("✗ Namespace 'vehiculos' NO encontrado en 'usa'")
            print("  Namespaces disponibles en 'usa':")
            for ns in usa_resolver.namespace_dict.keys():
                print(f"    - {ns}")
    else:
        print("✗ Namespace 'usa' NO encontrado")
except Exception as e:
    print(f"Error: {e}")
    import traceback
    traceback.print_exc()
exit()
