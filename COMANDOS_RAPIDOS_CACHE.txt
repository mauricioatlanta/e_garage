🚨 PROBLEMA: Botones no se actualizan en el servidor

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

📋 PASO 1: Subir scripts al servidor

Sube estos archivos con FileZilla:
- FIX_CACHE_SERVIDOR.sh
- VERIFICAR_ARCHIVOS_SERVIDOR.sh

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

🖥️  PASO 2: Ejecutar en el servidor (SSH)

cd /ruta/a/egarage

# Dar permisos
chmod +x FIX_CACHE_SERVIDOR.sh VERIFICAR_ARCHIVOS_SERVIDOR.sh

# Verificar archivos primero
./VERIFICAR_ARCHIVOS_SERVIDOR.sh

# Si los archivos están correctos, limpiar cache
./FIX_CACHE_SERVIDOR.sh

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

🔄 PASO 3: Reiniciar servidor

PythonAnywhere:
- Web tab → Reload

Otros:
sudo systemctl restart gunicorn-egarage

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

🌐 PASO 4: Limpiar cache del navegador

Chrome/Edge: Ctrl+Shift+Delete → Caché → Borrar
O usa modo incógnito: Ctrl+Shift+N

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

✅ VERIFICAR:

Abrir en modo incógnito:
https://www.egarage.cl/us/es/bienvenida/

Debe mostrar:
✅ Botones con bordes neon brillantes
✅ Espaciado aumentado (gap-5)
✅ Estilos neon-glow aplicados
