import sys
import os

# Verificar que el archivo colombia.py existe
colombia_path = "taller/urls_extra/colombia.py"
if os.path.exists(colombia_path):
    print(f"✅ Archivo {colombia_path} existe")
    with open(colombia_path, 'r', encoding='utf-8') as f:
        content = f.read()
        if 'app_name = "colombia"' in content:
            print("✅ app_name está definido correctamente")
        else:
            print("❌ app_name NO está definido")
        if 'urlpatterns = [' in content:
            print("✅ urlpatterns está definido")
        else:
            print("❌ urlpatterns NO está definido")
        if 'HttpResponseRedirect("/co/egarage/")' in content or 'HttpResponseRedirect("/co/")' in content:
            print("✅ Redirección de inicio está definida")
        else:
            print("❌ Redirección de inicio NO está definida")
else:
    print(f"❌ Archivo {colombia_path} NO existe")

# Verificar que las rutas están en gestion_taller/urls.py
urls_path = "gestion_taller/urls.py"
if os.path.exists(urls_path):
    print(f"\n✅ Archivo {urls_path} existe")
    with open(urls_path, 'r', encoding='utf-8') as f:
        content = f.read()
        if 'path("co/",' in content:
            print("✅ Ruta 'co/' está definida en gestion_taller/urls.py")
        else:
            print("❌ Ruta 'co/' NO está definida en gestion_taller/urls.py")
        if 'taller.urls_extra.colombia' in content:
            print("✅ Importación de colombia está en gestion_taller/urls.py")
        else:
            print("❌ Importación de colombia NO está en gestion_taller/urls.py")
else:
    print(f"❌ Archivo {urls_path} NO existe")

# Intentar importar el módulo para verificar errores de sintaxis
try:
    sys.path.insert(0, os.getcwd())
    from taller.urls_extra import colombia
    print("\n✅ Módulo colombia.py se puede importar correctamente")
    print(f"✅ app_name: {colombia.app_name}")
    print(f"✅ Número de URL patterns: {len(colombia.urlpatterns)}")
except Exception as e:
    print(f"\n❌ Error al importar colombia.py: {e}")
    import traceback
    traceback.print_exc()

PYEOF
