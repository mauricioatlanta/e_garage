import os
import django
from django.test import RequestFactory

os.environ.setdefault('DJANGO_SETTINGS_MODULE', 'gestion_taller.settings')

try:
    django.setup()
except Exception as e:
    print(f"Warning: {e}")

from taller.models import Empresa
from taller.servicios.models import (
    CategoriaServicio, CategoriaServicioName,
    Servicio, ServicioName,
    SubcategoriaServicio, SubcategoriaServicioName,
)

country_code = "US"
language = "en"

# 1. Verificar datos en BD
print(f"\n📊 DATOS EN BASE DE DATOS:")
servicios_us_bd = Servicio.objects.filter(categoria__country=country_code)
print(f"   Servicios con categoría US: {servicios_us_bd.count()}")

servicios_con_en = servicios_us_bd.filter(
    names__language=language,
    names__is_default=True
).distinct()
print(f"   Servicios con nombre EN: {servicios_con_en.count()}")

categorias_us = CategoriaServicio.objects.filter(country=country_code)
print(f"   Categorías US: {categorias_us.count()}")

for cat in categorias_us:
    nombre = cat.get_label(language)
    servicios_cat = servicios_con_en.filter(categoria=cat).count()
    print(f"   - {cat.code}: {nombre or 'SIN NOMBRE'} ({servicios_cat} servicios)")

# 2. Simular la vista como usuario anónimo
print(f"\n🔍 SIMULANDO VISTA (Usuario Anónimo):")
factory = RequestFactory()
request = factory.get('/us/servicios/')
request.user = AnonymousUser()
request.LANGUAGE_CODE = 'en'

try:
    # Simular detección de país
    request.path = '/us/servicios/'
    
    # Ejecutar la lógica de la vista manualmente
    empresa = getattr(request.user, "empresa", None)
    print(f"   Empresa del usuario: {empresa}")
    
    # Detectar país
    country_code_sim = "US"  # Porque la URL es /us/
    lang_sim = "en"
    language_sim = "en"
    
    print(f"   País detectado: {country_code_sim}")
    print(f"   Idioma: {language_sim}")
    
    # Filtrar servicios
    servicios = Servicio.objects.filter(categoria__country=country_code_sim)
    print(f"   Servicios filtrados por país: {servicios.count()}")
    
    servicios_qs = servicios.select_related("categoria", "subcategoria").prefetch_related(
        "names", "categoria__names", "subcategoria__names"
    ).order_by("nombre")
    
    servicios_qs = servicios_qs.filter(names__language=language_sim).distinct()
    servicios_list = list(servicios_qs)
    print(f"   Servicios después de filtrar por idioma: {len(servicios_list)}")
    
    # Verificar categorías
    categorias_qs = CategoriaServicio.objects.filter(country=country_code_sim).prefetch_related("names").order_by("code").distinct()
    categorias = []
    for categoria in categorias_qs:
        nombre = categoria.get_label(language_sim)
        if nombre and nombre != categoria.code:
            categorias.append(categoria)
    print(f"   Categorías válidas: {len(categorias)}")
    
    # Verificar servicios_por_categoria
    from collections import defaultdict
    servicios_por_categoria = defaultdict(list)
    for servicio in servicios_list:
        if servicio.categoria and servicio.categoria.country == country_code_sim:
            servicios_por_categoria[servicio.categoria].append(servicio)
    
    print(f"   Categorías con servicios: {len(servicios_por_categoria)}")
    for cat, servs in servicios_por_categoria.items():
        print(f"     - {cat.code}: {len(servs)} servicios")
    
except Exception as e:
    print(f"   ❌ Error: {e}")
    import traceback
    traceback.print_exc()
