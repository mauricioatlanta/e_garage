from django.contrib.auth import get_user_model
User = get_user_model()

try:
    user = User.objects.get(username="testuser_usa")
    print(f"Usuario: {user.username}")
    print(f"Email: {user.email}")
    print(f"Activo: {user.is_active}")
    print(f"Staff: {user.is_staff}")
    print(f"Último login: {user.last_login}")
    print(f"Fecha registro: {user.date_joined}")
    
    # Probar contraseña
    from django.contrib.auth import authenticate
    auth_user = authenticate(username="testuser_usa", password="TestUSA2025!")
    if auth_user:
        print("✅ Autenticación exitosa")
    else:
        print("❌ Autenticación fallida")
        # Resetear contraseña
        user.set_password("TestUSA2025!")
        user.is_active = True
        user.save()
        print("✅ Contraseña reseteada")
        
        # Verificar de nuevo
        auth_user = authenticate(username="testuser_usa", password="TestUSA2025!")
        if auth_user:
            print("✅ Ahora la autenticación funciona")
        else:
            print("❌ ERROR: Aún no funciona")
except User.DoesNotExist:
    print("❌ Usuario no existe")
