from pathlib import Path

# 1. Corregir template PDF
file_path = 'templates/taller/documentos/pdf_template.html'

print(f"📝 Corrigiendo {file_path}...\n")

if not Path(file_path).exists():
    print(f"❌ Error: {file_path} no existe")
    exit(1)

with open(file_path, 'r', encoding='utf-8') as f:
    content = f.read()

# Corregir nombre_servicio a nombre
if 'servicio.nombre_servicio' in content:
    content = content.replace('servicio.nombre_servicio', 'servicio.nombre')
    print("✅ Campo nombre_servicio corregido a nombre")

# Mover totales después de Servicios Subcontratados
# Buscar el cierre de la sección de otros_servicios
if '<!-- SEGMENTO OBSERVACIONES -->' in content and '<!-- TOTALES AL PIE -->' not in content:
    # Los totales ya están en el lugar correcto
    print("ℹ️  Los totales ya están en el lugar correcto")
elif '<!-- TOTALES AL PIE -->' in content:
    # Verificar que estén después de otros_servicios
    idx_otros = content.find('{% endif %}', content.find('Servicios Subcontratados'))
    idx_totales = content.find('<!-- TOTALES AL PIE -->')
    if idx_totales > idx_otros:
        print("✅ Los totales están después de Servicios Subcontratados")
    else:
        print("⚠️  Los totales necesitan ser movidos manualmente")

# Guardar
with open(file_path, 'w', encoding='utf-8') as f:
    f.write(content)

print(f"\n✅ Archivo {file_path} actualizado")

# 2. Corregir exporter
file_path2 = 'taller/utils/export_utils.py'

print(f"\n📝 Corrigiendo {file_path2}...\n")

if not Path(file_path2).exists():
    print(f"❌ Error: {file_path2} no existe")
    exit(1)

with open(file_path2, 'r', encoding='utf-8') as f:
    content2 = f.read()

# Corregir relaciones
if 'self.documento.repuestos.all()' in content2:
    content2 = content2.replace('self.documento.repuestos.all()', 'self.documento.lineas_repuesto.all()')
    print("✅ Relación repuestos corregida")
if 'self.documento.servicios.all()' in content2:
    content2 = content2.replace('self.documento.servicios.all()', 'self.documento.lineas_servicio.all()')
    print("✅ Relación servicios corregida")
if 'self.documento.otros_servicios.all()' in content2:
    content2 = content2.replace('self.documento.otros_servicios.all()', 'self.documento.lineas_otro_servicio.all()')
    print("✅ Relación otros_servicios corregida")

# Guardar
with open(file_path2, 'w', encoding='utf-8') as f:
    f.write(content2)

print(f"\n✅ Archivo {file_path2} actualizado")
PYEOF
