import os
import re

file_path = "taller/documentos/views.py"

with open(file_path, 'r', encoding='utf-8') as f:
    content = f.read()

# Crear backup
with open(f"{file_path}.backup", 'w', encoding='utf-8') as f:
    f.write(content)

# Buscar y corregir el f-string problemático
# Buscar línea con: f"https://wa.me/{telefono}?text={mensaje.replace(...).replace('\n', ...)}"
lines = content.split('\n')
new_lines = []
i = 0
found = False

while i < len(lines):
    line = lines[i]
    # Buscar línea con wa.me y mensaje.replace con \n
    if 'wa.me' in line and 'mensaje' in line and ('replace' in line or '\\n' in line) and 'f"' in line:
        # Encontrar indentación
        indent = len(line) - len(line.lstrip())
        # Reemplazar con código corregido
        new_lines.append(' ' * indent + '# Crear URL de WhatsApp')
        new_lines.append(' ' * indent + '# Nota: No se pueden usar backslashes directamente en expresiones f-string')
        new_lines.append(' ' * indent + 'mensaje_encoded = mensaje.replace(" ", "%20").replace("\\n", "%0A")')
        new_lines.append(' ' * indent + 'url_whatsapp = f"https://wa.me/{telefono}?text={mensaje_encoded}"')
        found = True
        print(f"✅ Línea {i+1} corregida")
    else:
        new_lines.append(line)
    i += 1

if found:
    new_content = '\n'.join(new_lines)
    with open(file_path, 'w', encoding='utf-8') as f:
        f.write(new_content)
    
    # Verificar sintaxis
    compile(new_content, file_path, 'exec')
    print("✅ Sintaxis válida")
else:
    print("⚠️  No se encontró el f-string problemático")
