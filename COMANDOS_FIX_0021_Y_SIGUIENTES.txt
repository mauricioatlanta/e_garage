SCRIPT AUTOMATIZADO (Si hay muchos errores):

# Crear script que maneje errores automáticamente
cat > /tmp/auto_fix_migrations.sh << 'EOF'
#!/bin/bash
cd /home/atlantareciclajes/apps/egarage/current
workon venv_egarage310

MAX_ATTEMPTS=10
ATTEMPT=0

while [ $ATTEMPT -lt $MAX_ATTEMPTS ]; do
    echo "Intento $((ATTEMPT + 1)): Aplicando migraciones..."
    OUTPUT=$(python manage.py migrate 2>&1)
    echo "$OUTPUT"
    
    if echo "$OUTPUT" | grep -q "already exists"; then
        # Extraer nombre de migración
        MIG=$(echo "$OUTPUT" | grep -oP "taller\.\K[0-9]+" | head -1)
        if [ -n "$MIG" ]; then
            echo "Marcando migración $MIG como aplicada (fake)..."
            python manage.py migrate taller $MIG --fake
            ATTEMPT=$((ATTEMPT + 1))
        else
            echo "No se pudo determinar la migración. Revisa manualmente."
            break
        fi
    elif echo "$OUTPUT" | grep -q "No migrations to apply"; then
        echo "✅ Todas las migraciones aplicadas correctamente"
        break
    else
        echo "✅ Migraciones aplicadas sin errores"
        break
    fi
done
EOF

chmod +x /tmp/auto_fix_migrations.sh
