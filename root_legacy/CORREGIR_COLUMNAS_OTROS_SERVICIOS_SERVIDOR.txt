from pathlib import Path

file_path = 'templates/taller/common/documentos/ver_documento_nuevo.html'

print(f"📝 Corrigiendo {file_path}...\n")

if not Path(file_path).exists():
    print(f"❌ Error: {file_path} no existe")
    exit(1)

with open(file_path, 'r', encoding='utf-8') as f:
    content = f.read()

# Intercambiar los campos en la tabla de otros servicios
# Buscar el patrón donde se muestran los campos
old_pattern = r'<td class="py-3 px-4">{{ otro\.nombre\|default:"N/A" }}</td>\s+<td class="py-3 px-4">{{ otro\.empresa_externa\|default:"N/A" }}</td>'
new_pattern = r'<td class="py-3 px-4">{{ otro.empresa_externa|default:"N/A" }}</td>\n              <td class="py-3 px-4">{{ otro.nombre|default:"N/A" }}</td>'

if '{{ otro.nombre|default:"N/A" }}</td>' in content and '{{ otro.empresa_externa|default:"N/A" }}</td>' in content:
    # Intercambiar usando reemplazo directo
    import re
    # Buscar la línea específica y reemplazar
    lines = content.split('\n')
    new_lines = []
    i = 0
    while i < len(lines):
        line = lines[i]
        # Si encontramos la línea con otro.nombre seguida de otra con otro.empresa_externa
        if '{{ otro.nombre|default:"N/A" }}</td>' in line:
            # Buscar la siguiente línea con empresa_externa
            if i + 1 < len(lines) and '{{ otro.empresa_externa|default:"N/A" }}</td>' in lines[i + 1]:
                # Intercambiar
                new_lines.append(lines[i + 1])  # Primero empresa_externa
                new_lines.append(line)  # Luego nombre
                i += 2
                continue
        new_lines.append(line)
        i += 1
    
    content = '\n'.join(new_lines)
    print("✅ Campos intercambiados en la tabla de otros servicios")
else:
    print("ℹ️  No se encontraron los campos para intercambiar")

# Guardar

    f.write(content)
