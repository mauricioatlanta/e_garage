# Crear script que marque migraciones como fake cuando falle
cat > /tmp/fix_migrations.sh << 'EOF'
#!/bin/bash
cd /home/atlantareciclajes/apps/egarage/current
workon venv_egarage310

while true; do
    python manage.py migrate 2>&1 | tee /tmp/migrate_output.log
    
    if grep -q "already exists" /tmp/migrate_output.log; then
        # Extraer nombre de la migración que falló
        MIGRATION=$(grep "already exists" /tmp/migrate_output.log | head -1 | grep -o "taller\.[0-9]*" | head -1)
        if [ -n "$MIGRATION" ]; then
            echo "Marcando $MIGRATION como aplicada (fake)..."
            python manage.py migrate taller ${MIGRATION##*.} --fake
        else
            echo "No se pudo determinar la migración. Revisa manualmente."
            break
        fi
    else
        echo "✅ Todas las migraciones aplicadas correctamente"
        break
    fi
done
EOF

chmod +x /tmp/fix_migrations.sh
