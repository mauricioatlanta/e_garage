===========================================
CORREGIR APP_NAMESPACE EN TEMPLATE
===========================================

PROBLEMA:
---------
El tag country_url está interpretando app_namespace='direct' como namespace literal
Necesitamos usar app_namespace="" (vacío) en lugar de 'direct'

SOLUCIÓN:
---------

cd /home/atlantareciclajes/apps/egarage/current
workon venv_egarage310

# Reemplazar app_namespace='direct' por app_namespace=""
sed -i "s/app_namespace='direct'/app_namespace=\"\"/g" \
   taller/templates/taller/us/en/vehiculos/crear_vehiculo.html

# Verificar que se cambió
grep -n "app_namespace" taller/templates/taller/us/en/vehiculos/crear_vehiculo.html

# Verificar las URLs corregidas
grep -A 1 -B 1 "modelos_por_marca_api" taller/templates/taller/us/en/vehiculos/crear_vehiculo.html

# Limpiar caché
find . -type d -name "__pycache__" -exec rm -r {} + 2>/dev/null
find . -name "*.pyc" -delete

# Verificar que el template se puede renderizar
python manage.py shell << 'PYTHON'
from django.template.loader import get_template
from django.contrib.auth.models import User
from django.test import RequestFactory
from taller.vehiculos.forms import VehiculoForm

factory = RequestFactory()
request = factory.get('/us/vehiculos/crear/')
user = User.objects.get(username='testuser_usa')
request.user = user

try:
    from taller.vehiculos.views_fbv import _vehicle_template
    template_name = _vehicle_template("crear", "US")
    template = get_template(template_name)
    
    form = VehiculoForm(user=user, request=request)
    html = template.render({
        'form': form,
        'country': 'US',
        'request': request
    })
    
    print("✓ Template renderizado correctamente")
    print(f"  Tamaño HTML: {len(html)} caracteres")
    
    if 'NoReverseMatch' in html or 'Reverse for' in html:
        print("✗ Aún hay errores de URL")
    else:
        print("✓ No hay errores de URL")
        
except Exception as e:
    print(f"✗ Error: {e}")
    import traceback
    traceback.print_exc()
exit()
