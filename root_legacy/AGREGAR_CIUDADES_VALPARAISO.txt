import os
import django
os.environ.setdefault('DJANGO_SETTINGS_MODULE', 'gestion_taller.settings')
django.setup()

from taller.models.region_ciudad import TallerRegion, TallerCiudad

# Ciudades principales de Valparaíso
ciudades_valparaiso = [
    "Valparaíso",
    "Viña del Mar",
    "Quilpué",
    "Villa Alemana",
    "San Antonio",
    "Quillota",
    "La Calera",
    "Los Andes",
    "San Felipe",
    "La Ligua",
    "Petorca",
    "Limache",
    "Olmué",
    "Casablanca",
    "Algarrobo",
    "El Quisco",
    "El Tabo",
    "Cartagena",
    "Santo Domingo",
    "Concón",
]

print("🔧 Agregando ciudades a Valparaíso...\n")
print("=" * 70)

# Buscar región (puede estar con o sin acento)
region = TallerRegion.objects.filter(nombre__iexact="Valparaíso").first()
if not region:
    region = TallerRegion.objects.filter(nombre__iexact="Valparaiso").first()

if not region:
    print("❌ No se encontró la región Valparaíso")
    print("💡 Creando la región primero...")
    region = TallerRegion.objects.create(nombre="Valparaíso")
    print(f"✅ Región creada: {region.nombre}")
else:
    print(f"✅ Región encontrada: {region.nombre} (ID: {region.id})")

print(f"\n📊 Ciudades actuales en {region.nombre}: {TallerCiudad.objects.filter(region=region).count()}\n")

ciudades_creadas = 0
ciudades_existentes = 0

for ciudad_nombre in ciudades_valparaiso:
    ciudad, ciudad_creada = TallerCiudad.objects.get_or_create(
        nombre=ciudad_nombre,
        region=region
    )
    if ciudad_creada:
        print(f"   ✅ Ciudad creada: {ciudad_nombre}")
        ciudades_creadas += 1
    else:
        print(f"   ℹ️  Ciudad existente: {ciudad_nombre}")
        ciudades_existentes += 1

print()
print("=" * 70)
print(f"📊 Resumen para {region.nombre}:")
print(f"   ✅ Ciudades creadas: {ciudades_creadas}")
print(f"   ℹ️  Ciudades existentes: {ciudades_existentes}")
print(f"   📈 Total ciudades en la región: {TallerCiudad.objects.filter(region=region).count()}")
print("=" * 70)
print("\n✅ Proceso completado")
