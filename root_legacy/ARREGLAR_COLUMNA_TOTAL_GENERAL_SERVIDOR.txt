import os
import django
import sqlite3
from pathlib import Path

os.environ.setdefault('DJANGO_SETTINGS_MODULE', 'gestion_taller.settings')
django.setup()

from django.conf import settings

# Obtener la ruta de la base de datos
db_path = settings.DATABASES['default']['NAME']
print(f"📁 Base de datos: {db_path}")

# Conectar a SQLite
conn = sqlite3.connect(db_path)
cursor = conn.cursor()

# Verificar si la columna existe
cursor.execute("PRAGMA table_info(taller_documento)")
columns = [row[1] for row in cursor.fetchall()]

if 'total_general' in columns:
    print("✅ La columna 'total_general' ya existe")
else:
    print("⚠️  La columna 'total_general' NO existe. Agregándola...")
    
    try:
        # Agregar la columna
        cursor.execute("""
            ALTER TABLE taller_documento 
            ADD COLUMN total_general DECIMAL(14, 2) DEFAULT 0.00
        """)
        conn.commit()
        print("✅ Columna 'total_general' agregada exitosamente")
        
        # Actualizar valores existentes a 0.00 si es necesario
        cursor.execute("""
            UPDATE taller_documento 
            SET total_general = 0.00 
            WHERE total_general IS NULL
        """)
        conn.commit()
        print("✅ Valores inicializados")
        
    except sqlite3.Error as e:
        print(f"❌ Error: {e}")
        conn.rollback()

conn.close()
print("\n✅ Proceso completado")
PYEOF

cd /home/atlantareciclajes/apps/egarage/current && \
python3 << 'PYEOF'
import os
import django
import sqlite3
from pathlib import Path

os.environ.setdefault('DJANGO_SETTINGS_MODULE', 'gestion_taller.settings')
django.setup()

from django.conf import settings

db_path = settings.DATABASES['default']['NAME']
print(f"📁 Base de datos: {db_path}\n")

conn = sqlite3.connect(db_path)
cursor = conn.cursor()

# Columnas que deberían existir según la migración 0036
required_columns = {
    'total_general': 'DECIMAL(14, 2) DEFAULT 0.00',
    'total_repuestos': 'DECIMAL(14, 2) DEFAULT 0.00',
    'total_servicios': 'DECIMAL(14, 2) DEFAULT 0.00',
    'iva': 'DECIMAL(14, 2) DEFAULT 0.00',
}

# Verificar columnas existentes
cursor.execute("PRAGMA table_info(taller_documento)")
existing_columns = {row[1]: row[2] for row in cursor.fetchall()}

print("🔍 Verificando columnas...\n")

for col_name, col_type in required_columns.items():
    if col_name in existing_columns:
        print(f"✅ {col_name}: existe")
    else:
        print(f"⚠️  {col_name}: NO existe. Agregando...")
        try:
            cursor.execute(f"""
                ALTER TABLE taller_documento 
                ADD COLUMN {col_name} {col_type}
            """)
            conn.commit()
            print(f"   ✅ {col_name} agregada")
        except sqlite3.Error as e:
            print(f"   ❌ Error agregando {col_name}: {e}")

conn.close()
print("\n✅ Verificación completada")
PYEOF

touch /var/www/www_egarage_cl_wsgi.py
