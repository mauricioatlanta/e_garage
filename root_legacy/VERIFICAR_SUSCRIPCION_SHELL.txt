========================================
CÓDIGO PARA VERIFICAR SUSCRIPCIÓN
========================================

En el shell de Django, ejecuta esto:

# Importar desde el módulo correcto
from taller.models.suscripcion import Suscripcion
from django.contrib.auth import get_user_model
from datetime import datetime, timedelta

User = get_user_model()
user = User.objects.get(username="testuser_usa")

# Verificar si tiene suscripción
try:
    suscripcion = user.suscripcion
    print(f"✅ Suscripción encontrada:")
    print(f"   Tipo: {suscripcion.tipo}")
    print(f"   Activa: {suscripcion.activa}")
    print(f"   Fecha inicio: {suscripcion.fecha_inicio}")
    print(f"   Fecha fin: {suscripcion.fecha_fin}")
    
    # Renovar si está inactiva o vencida
    if not suscripcion.activa or (suscripcion.fecha_fin and suscripcion.fecha_fin < datetime.now().date()):
        print("⚠️  Suscripción inactiva o vencida. Renovando...")
        suscripcion.activa = True
        suscripcion.fecha_inicio = datetime.now().date()
        suscripcion.fecha_fin = (datetime.now() + timedelta(days=30)).date()
        suscripcion.save()
        print(f"✅ Suscripción renovada hasta {suscripcion.fecha_fin}")
    else:
        print(f"✅ Suscripción vigente hasta {suscripcion.fecha_fin}")
        
except AttributeError:
    print("❌ No tiene suscripción. Creando...")
    suscripcion = Suscripcion.objects.create(
        user=user,
        tipo='trial',
        fecha_inicio=datetime.now().date(),
        fecha_fin=(datetime.now() + timedelta(days=30)).date(),
        activa=True
    )
    print(f"✅ Suscripción creada hasta {suscripcion.fecha_fin}")
