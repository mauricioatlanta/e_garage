import os
import django
os.environ.setdefault('DJANGO_SETTINGS_MODULE', 'gestion_taller.settings')
django.setup()

from taller.models.region_ciudad import TallerRegion, TallerCiudad

print("🔍 Verificando regiones y ciudades de Chile:\n")

# Verificar regiones
regiones = TallerRegion.objects.all()
print(f"📊 Total de regiones: {regiones.count()}")
if regiones.count() > 0:
    print("\n✅ Regiones disponibles:")
    for r in regiones[:10]:  # Mostrar primeras 10
        ciudades_count = TallerCiudad.objects.filter(region=r).count()
        print(f"   - {r.id}: {r.nombre} ({ciudades_count} ciudades)")
else:
    print("❌ No hay regiones en la base de datos")
    print("💡 Necesitas crear regiones primero")

# Verificar ciudades
ciudades = TallerCiudad.objects.all()
print(f"\n📊 Total de ciudades: {ciudades.count()}")
if ciudades.count() > 0:
    print("\n✅ Primeras ciudades:")
    for c in ciudades[:5]:
        print(f"   - {c.id}: {c.nombre} (Región: {c.region.nombre if c.region else 'N/A'})")
else:
    print("❌ No hay ciudades en la base de datos")

print("\n" + "="*60)
print("💡 Si no hay regiones, necesitas crearlas primero")
print("="*60)
