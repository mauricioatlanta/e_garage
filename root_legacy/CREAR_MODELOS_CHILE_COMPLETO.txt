total_creados = 0

marcas_no_encontradas = []

for marca_nombre, modelos_lista in modelos_por_marca.items():
    try:
        marca = Marca.objects.get(nombre=marca_nombre, country='CL')
        print(f"\n🚗 Procesando marca: {marca_nombre}")
        
        marca_creados = 0
        marca_existentes = 0
        
        for modelo_nombre in modelos_lista:
            modelo, created = Modelo.objects.get_or_create(
                nombre=modelo_nombre,
                marca=marca,
                country='CL',
                defaults={'nombre': modelo_nombre, 'marca': marca, 'country': 'CL'}
            )
            
            if created:
                print(f"   ✅ {modelo_nombre}")
                marca_creados += 1
                total_creados += 1
            else:
                marca_existentes += 1
                total_existentes += 1
        
        print(f"   📊 {marca_nombre}: {marca_creados} creados, {marca_existentes} existentes")
        
    except Marca.DoesNotExist:
        print(f"⚠️  Marca no encontrada: {marca_nombre}")
        marcas_no_encontradas.append(marca_nombre)
        continue

print()

    print(f"   ⚠️  Marcas no encontradas: {', '.join(marcas_no_encontradas)}")
