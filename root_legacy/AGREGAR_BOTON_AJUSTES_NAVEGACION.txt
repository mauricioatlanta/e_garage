import re
from pathlib import Path

files_to_update = [
    'templates/common/base.html',
    'templates/base.html',
    'templates/taller/common/base.html'
]

print("📝 Actualizando archivos de navegación...\n")

for file_path in files_to_update:
    if not Path(file_path).exists():
        print(f"⚠️  {file_path} no existe, saltando...")
        continue
    
    print(f"\n📄 Procesando {file_path}...")
    
    with open(file_path, 'r', encoding='utf-8') as f:
        content = f.read()
    
    original_content = content
    
    # Buscar el patrón: botón de VEHÍCULOS seguido de CERRAR SESIÓN/SALIR
    # Patrón 1: VEHÍCULOS seguido de form con logout
    pattern1 = r'(<a href="[^"]*vehiculos[^"]*"[^>]*>.*?VEHÍCULOS.*?</a>\s*)'
    pattern2 = r'(<a href="[^"]*vehiculos[^"]*"[^>]*>.*?VEHICLES.*?</a>\s*)'
    
    new_button = '''      <a href="/cl/es/settings/" class="nav-button-standard">
        <span class="nav-icon">⚙️</span>
        <span>{% if LANGUAGE_CODE == 'es' %}AJUSTES{% else %}SETTINGS{% endif %}</span>
      </a>

      '''
    
    # Verificar si ya existe el botón de Ajustes después de Vehículos
    if 'VEHÍCULOS' in content or 'VEHICLES' in content:
        # Buscar si ya hay un botón de Ajustes después de Vehículos
        if re.search(r'vehiculos.*?ajustes|vehicles.*?settings', content, re.IGNORECASE | re.DOTALL):
            print(f"   ℹ️  El botón 'Ajustes' ya existe después de 'Vehículos'")
        else:
            # Buscar el patrón: VEHÍCULOS seguido de form/logout
            vehiculos_pattern = r'(<a href="[^"]*vehiculos[^"]*"[^>]*>.*?(?:VEHÍCULOS|VEHICLES).*?</a>\s*)(\s*<form[^>]*logout|</form>|CERRAR|SALIR|LOGOUT)'
            
            if re.search(vehiculos_pattern, content, re.IGNORECASE | re.DOTALL):
                # Reemplazar agregando el botón de Ajustes
                replacement = r'\1' + new_button + r'\2'
                content = re.sub(vehiculos_pattern, replacement, content, flags=re.IGNORECASE | re.DOTALL)
                print(f"   ✅ Botón 'Ajustes' agregado después de 'Vehículos'")
            else:
                # Buscar solo el botón de Vehículos y agregar después
                vehiculos_simple = r'(<a href="[^"]*vehiculos[^"]*"[^>]*>.*?(?:VEHÍCULOS|VEHICLES).*?</a>\s*)'
                if re.search(vehiculos_simple, content, re.IGNORECASE | re.DOTALL):
                    content = re.sub(vehiculos_simple, r'\1' + new_button, content, flags=re.IGNORECASE | re.DOTALL)
                    print(f"   ✅ Botón 'Ajustes' agregado después de 'Vehículos' (patrón simple)")
                else:
                    print(f"   ⚠️  No se encontró el patrón de 'Vehículos'")
    
    # Si hubo cambios, guardar
    if content != original_content:
        with open(file_path, 'w', encoding='utf-8') as f:
            f.write(content)
        print(f"   💾 {file_path} guardado")
    else:
        print(f"   ℹ️  {file_path} sin cambios necesarios")

print("\n" + "="*70)
print("✅ PROCESO COMPLETADO")
print("="*70)
