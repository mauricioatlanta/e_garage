import re

file_path = "templates/taller/us/en/dashboard/centro_operaciones_espacial.html"

with open(file_path, 'r', encoding='utf-8') as f:
    content = f.read()

# Verificar si tiene {% extends %}
has_extends = '{% extends' in content

# Contar bloques
block_count = len(re.findall(r'{%\s*block\s+', content))
endblock_count = len(re.findall(r'{%\s*endblock', content))

print(f"📊 Estado inicial: {block_count} blocks, {endblock_count} endblocks, extends={has_extends}")

# Si no tiene extends O los bloques están desbalanceados, eliminar TODOS
if not has_extends or block_count != endblock_count:
    print("⚠️  Eliminando TODOS los bloques...")
    lines = content.splitlines(keepends=True)
    new_lines = []
    removed = 0
    
    for i, line in enumerate(lines, 1):
        if '{% block' in line or '{% endblock' in line:
            print(f"   Línea {i}: {line.strip()[:60]}")
            removed += 1
            continue
        new_lines.append(line)
    
    content = ''.join(new_lines)
    print(f"✅ Eliminadas {removed} líneas")

# Escribir archivo corregido
with open(file_path, 'w', encoding='utf-8') as f:
    f.write(content)

print("✅ Archivo corregido")
PYEOF

touch /var/www/www_egarage_cl_wsgi.py && \
echo "✅ Servidor reiniciado"
