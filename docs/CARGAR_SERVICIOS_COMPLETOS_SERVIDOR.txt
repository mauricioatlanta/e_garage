empresas = Empresa.objects.all()
print(f"\n📋 Empresas encontradas: {empresas.count()}")

servicios_creados = 0
servicios_actualizados = 0

with transaction.atomic():
    for country, lang, label_key in [('CL', 'es', 'es'), ('US', 'en', 'en')]:
        print(f"\n🌍 Procesando servicios para {country} ({lang})")
        
        empresas_pais = empresas.filter(pais=country)
        if not empresas_pais.exists():
            print(f"   ⚠️ No hay empresas con país {country}")
            continue
        
        for cat in categorias_completas:
            cat_obj, _ = CategoriaServicio.objects.get_or_create(country=country, code=cat['code'])
            CategoriaServicioName.objects.get_or_create(
                categoria=cat_obj, language=lang, is_default=True,
                defaults={'label': cat[label_key]}
            )
            
            for sub in cat['subcategorias']:
                sub_obj, _ = SubcategoriaServicio.objects.get_or_create(
                    categoria=cat_obj, code=sub['code'], country=country
                )
                SubcategoriaServicioName.objects.get_or_create(
                    subcategoria=sub_obj, language=lang, is_default=True,
                    defaults={'label': sub[label_key]}
                )
                
                for serv in sub['servicios']:
                    for empresa in empresas_pais:
                        # Buscar servicio existente
                        try:
                            servicio = Servicio.objects.get(
                                empresa=empresa,
                                categoria=cat_obj,
                                nombre=serv[label_key]
                            )
                            if servicio.subcategoria != sub_obj:
                                servicio.subcategoria = sub_obj
                                servicio.save()
                            servicios_actualizados += 1
                        except Servicio.DoesNotExist:
                            # Crear nuevo servicio
                            try:
                                servicio = Servicio.objects.create(
                                    empresa=empresa,
                                    categoria=cat_obj,
                                    subcategoria=sub_obj,
                                    nombre=serv[label_key]
                                )
                                servicios_creados += 1
                            except IntegrityError:
                                try:
                                    servicio = Servicio.objects.get(
                                        empresa=empresa,
                                        categoria=cat_obj,
                                        nombre=serv[label_key]
                                    )
                                    servicios_actualizados += 1
                                except:
                                    continue
                        
                        # Asegurar que tiene nombre en el idioma correcto
                        ServicioName.objects.get_or_create(
                            servicio=servicio,
                            language=lang,
                            is_default=True,
                            defaults={'label': serv[label_key]}
                        )

print(f'\n✅ ¡Carga completada!')
print(f'📊 Servicios creados: {servicios_creados}')
print(f'📊 Servicios actualizados: {servicios_actualizados}')

# Verificación final
for country in ['CL', 'US']:
    servicios_finales = Servicio.objects.filter(
        categoria__country=country
    ).distinct()
    print(f'📊 Total servicios {country}: {servicios_finales.count()}')

for emp in empresas:
    servicios_emp = Servicio.objects.filter(empresa=emp)
    nombre = emp.nombre_taller or emp.empresa or f"Empresa {emp.id}"
    print(f'   - {nombre} ({emp.pais}): {servicios_emp.count()} servicios')
