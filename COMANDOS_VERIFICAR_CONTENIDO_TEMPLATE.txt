import os
template_path = "taller/templates/taller/us/en/vehiculos/crear_vehiculo.html"

if os.path.exists(template_path):
    with open(template_path, 'r', encoding='utf-8') as f:
        content = f.read()
        print(f"Tamaño del contenido: {len(content)} caracteres")
        print(f"Primeras 200 caracteres:")
        print(content[:200])
        print(f"\n¿Contiene 'Create Vehicle'? { 'Create Vehicle' in content }")
        print(f"¿Contiene 'form-card'? { 'form-card' in content }")
        print(f"¿Contiene '🌸'? { '🌸' in content }")
        print(f"¿Contiene '⭐'? { '⭐' in content }")
        
        # Buscar qué contiene realmente
        if '{% extends' in content:
            print("\n✓ Contiene {% extends")
        if 'base.html' in content:
            print("✓ Contiene base.html")
        if 'vehiculo_form.html' in content:
            print("✓ Contiene vehiculo_form.html")
            
        # Ver cuántas líneas tiene
        lines = content.split('\n')
        print(f"\nTotal de líneas: {len(lines)}")
        if len(lines) < 10:
            print("⚠️  ADVERTENCIA: Template muy corto, puede estar incompleto")
            print("Primeras líneas:")
            for i, line in enumerate(lines[:10], 1):
                print(f"  {i}: {line}")
else:
    print(f"✗ Archivo no existe: {template_path}")
exit()
