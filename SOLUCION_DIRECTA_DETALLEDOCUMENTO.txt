cd /home/atlantareciclajes/apps/egarage/current
workon venv_egarage310

# OPCIÓN 1: Obtener SQL y crear manualmente (RECOMENDADA)
# --------------------------------------------------------

# 1. Obtener el SQL para crear la tabla detalledocumento
python manage.py sqlmigrate taller 0001_initial_migration > /tmp/migration_0001.sql

# 2. Extraer solo la parte de detalledocumento
grep -A 50 'CREATE TABLE.*detalledocumento' /tmp/migration_0001.sql > /tmp/create_detalle.sql

# 3. Ver el contenido
cat /tmp/create_detalle.sql

# 4. Si el SQL se ve bien, ejecutarlo
python manage.py dbshell < /tmp/create_detalle.sql

# 5. Marcar migraciones como aplicadas
python manage.py migrate taller 0004 --fake

# 6. Continuar con migraciones
python manage.py migrate

# 7. Verificar
python manage.py check


# OPCIÓN 2: Revertir fake y aplicar de verdad (ALTERNATIVA)
# ----------------------------------------------------------

# 1. Desmarcar la migración 0001 (revertir el fake)
python manage.py migrate taller zero --fake

# 2. Aplicar la 0001 de verdad (creará solo las tablas que faltan)
#    Si falla con "table already exists", ignóralo y continúa
python manage.py migrate taller 0001 2>&1 | grep -v "already exists" || true

# 3. Aplicar el resto
python manage.py migrate


# OPCIÓN 3: Crear tabla con estructura básica (RÁPIDA)
# ------------------------------------------------------

python manage.py dbshell << 'EOF'
CREATE TABLE IF NOT EXISTS "taller_detalledocumento" (
    "id" integer NOT NULL PRIMARY KEY AUTOINCREMENT,
    "documento_id" bigint NOT NULL REFERENCES "taller_documento" ("id") DEFERRABLE INITIALLY DEFERRED,
    "servicio_id" bigint NULL REFERENCES "taller_servicio" ("id") DEFERRABLE INITIALLY DEFERRED,
    "repuesto_id" bigint NULL REFERENCES "taller_repuesto" ("id") DEFERRABLE INITIALLY DEFERRED,
    "cantidad" decimal NOT NULL,
    "precio_unitario" decimal NOT NULL,
    "descuento" decimal NOT NULL,
    "subtotal" decimal NOT NULL,
    "created_at" datetime NOT NULL,
    "updated_at" datetime NOT NULL
);
CREATE INDEX IF NOT EXISTS "taller_detalledocumento_documento_id_idx" ON "taller_detalledocumento" ("documento_id");
CREATE INDEX IF NOT EXISTS "taller_detalledocumento_servicio_id_idx" ON "taller_detalledocumento" ("servicio_id");
CREATE INDEX IF NOT EXISTS "taller_detalledocumento_repuesto_id_idx" ON "taller_detalledocumento" ("repuesto_id");
.quit
EOF

# Luego marcar migraciones como aplicadas
python manage.py migrate taller 0004 --fake
python manage.py migrate
